```python
def test_api_create_run_accepts_validate_phase(client, session, monkeypatch):
    """POST /api/runs with phase='validate' returns 200 and spawns the spider."""
    spawn_calls: list[dict] = []
    def fake_spawn(*, phase, shop, strategy="", mode="delta", urls="", cron_job_id=None):
        spawn_calls.append({
            "phase": phase, "shop": shop, "strategy": strategy,
            "mode": mode, "urls": urls, "cron_job_id": cron_job_id,
        })
    monkeypatch.setattr(
        "book_scraper.dashboard.routes.api._spawn_scrapy_in_container", fake_spawn
    )
    # Ensure a 'vaga' shop exists for the preflight check
    upsert_shop(session, "vaga", "https://www.vaga.lt")
    session.commit()

    resp = client.post("/api/runs", json={"shop": "vaga", "phase": "validate"})
    assert resp.status_code == 200, resp.text
    assert resp.json()["phase"] == "validate"
    assert len(spawn_calls) == 1
    assert spawn_calls[0]["phase"] == "validate"
    assert spawn_calls[0]["shop"] == "vaga"

def test_api_create_run_rejects_unknown_phase(client, session, monkeypatch):
    """POST /api/runs with phase='banana' still returns 400 (allowlist enforced)."""
    monkeypatch.setattr(
        "book_scraper.dashboard.routes.api._spawn_scrapy_in_container",
        lambda **kw: None,
    )
    upsert_shop(session, "vaga", "https://www.vaga.lt")
    session.commit()
    resp = client.post("/api/runs", json={"shop": "vaga", "phase": "banana"})
    assert resp.status_code == 400
    assert "Unknown phase" in resp.json()["detail"]

def test_api_create_run_rejects_concurrent_validate(client, session, monkeypatch):
    """A second validate run while the first is 'running' returns 409."""
    monkeypatch.setattr(
        "book_scraper.dashboard.routes.api._spawn_scrapy_in_container",
        lambda **kw: None,
    )
    shop = upsert_shop(session, "vaga", "https://www.vaga.lt")
    # Create an active validate run via repo helper.
    create_scrape_run(session, shop.id, "validate", extra_payload={"shop": "vaga"})
    session.commit()
    resp = client.post("/api/runs", json={"shop": "vaga", "phase": "validate"})
    assert resp.status_code == 409
```
